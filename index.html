const SUPABASE_URL = 'https://XXXXXXXXXXXXXX.supabase.co'; const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsIn...'; const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); // Semak sesi sedia ada semasa halaman dibuka window.addEventListener('DOMContentLoaded', async () => { try { const { data: { session }, error } = await supabase.auth.getSession(); if (error) throw error; if (session) { window.location.assign('dashboard.html'); } } catch (err) { console.error("Session check error:", err.message); } }); // Proses Log Masuk document.getElementById('loginForm').addEventListener('submit', async (e) => { e.preventDefault(); const email = document.getElementById('email').value.trim(); const password = document.getElementById('password').value; const btn = document.getElementById('btnLogin'); btn.textContent = "Sedang Log Masuk..."; btn.disabled = true; try { const { data, error } = await supabase.auth.signInWithPassword({ email: email, password: password }); if (error) { alert('Gagal Log Masuk: ' + error.message); btn.textContent = "Log Masuk"; btn.disabled = false; } else if (data.session) { // Guna window.location.assign untuk keserasian pelayar telefon yang lebih baik window.location.assign('dashboard.html'); } } catch (err) { alert('Ralat Sistem: ' + err.message); btn.textContent = "Log Masuk"; btn.disabled = false; } });
